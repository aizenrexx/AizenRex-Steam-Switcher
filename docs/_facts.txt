           from scratch, which was the bulk of the old start-up delay.
           R2R ships pre-compiled native code, so the window appears far sooner.
           TieredPGO then re-optimises the hot paths while the app runs. -->
      <PublishReadyToRun>true</PublishReadyToRun>
      <!-- Composite R2R bundles the whole framework into one image, which made
           the FIRST launch far slower (measured 39s) because Windows has to
           fault in a single huge image before anything renders. Per-assembly
           R2R keeps the pre-compiled code without that penalty. -->
      <PublishReadyToRunComposite>false</PublishReadyToRunComposite>
      <TieredCompilation>true</TieredCompilation>
      <TieredCompilationQuickStartup>true</TieredCompilationQuickStartup>

      <InvariantGlobalization>false</InvariantGlobalization>
